<script>

  const answers = { food: null, date: null, outfit: null };


  function goTo(id) {

    document.querySelectorAll('.scene').forEach(s => s.classList.remove('active'));

    document.getElementById(id).classList.add('active');

  }


  /* ---- Runaway NO button ---- */

  const noBtn = document.getElementById('noBtn');

  function dodge() {

    const pad = 20;

    const maxX = window.innerWidth - noBtn.offsetWidth - pad;

    const maxY = window.innerHeight - noBtn.offsetHeight - pad;

    const x = Math.max(pad, Math.random() * maxX);

    const y = Math.max(pad, Math.random() * maxY);

    noBtn.style.position = 'fixed';

    noBtn.style.left = x + 'px';

    noBtn.style.top = y + 'px';

  }

  noBtn.addEventListener('mouseenter', dodge);

  noBtn.addEventListener('click', dodge);

  noBtn.addEventListener('touchstart', function (e) { e.preventDefault(); dodge(); });


  /* ---- Choice selection (food & outfit) ---- */

  function wireChoice(choiceEl, containerId, key) {

    choiceEl.addEventListener('click', () => {

      const container = document.getElementById(containerId);

      container.querySelectorAll('.choice').forEach(x => x.classList.remove('selected'));

      choiceEl.classList.add('selected');

      answers[key] = choiceEl.dataset.value;

    });

  }

  const keyFor = { foodChoices: 'food', outfitChoices: 'outfit' };

  ['foodChoices', 'outfitChoices'].forEach(id => {

    document.getElementById(id).querySelectorAll('.choice')

      .forEach(c => wireChoice(c, id, keyFor[id]));

  });


  /* ---- Add a custom option ---- */

  function addChoice(containerId, inputId, emoji) {

    const input = document.getElementById(inputId);

    const text = input.value.trim();

    if (!text) { alert('Type something first! 😊'); return; }

    const container = document.getElementById(containerId);

    const div = document.createElement('div');

    div.className = 'choice';

    div.dataset.value = text + ' ' + emoji;

    div.innerHTML = '<span class="emoji">' + emoji + '</span>' + escapeHtml(text);

    container.appendChild(div);

    wireChoice(div, containerId, keyFor[containerId]);

    div.click(); // auto-select the new option

    input.value = '';

  }


  function escapeHtml(str) {

    return str.replace(/[&<>"']/g, m => ({

      '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'

    }[m]));

  }


  function nextFromFood() {

    if (!answers.food) { alert('Pick something yummy first! 😋'); return; }

    goTo('scene-date');

  }


  function nextFromDate() {

    const val = document.getElementById('dateInput').value;

    if (!val) { alert('Pick a day for our date! 📅'); return; }

    answers.date = val;

    goTo('scene-outfit');

  }


  function finish() {

    if (!answers.outfit) { alert('Pick an outfit vibe! 👗'); return; }

    document.getElementById('s-food').textContent = answers.food;

    document.getElementById('s-date').textContent = formatDate(answers.date);

    document.getElementById('s-outfit').textContent = answers.outfit;

    goTo('scene-done');

  }


  function formatDate(d) {

    const dt = new Date(d + 'T00:00:00');

    return dt.toLocaleDateString(undefined, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });

  }


  function restart() {

    answers.food = answers.date = answers.outfit = null;

    document.getElementById('dateInput').value = '';

    document.querySelectorAll('.choice').forEach(c => c.classList.remove('selected'));

    noBtn.style.position = 'relative';

    noBtn.style.left = noBtn.style.top = '';

    goTo('scene-anniv');

  }


  /* ---- Floating hearts background ---- */

  const hearts = ['💕', '💖', '💗', '🌸', '💘'];

  setInterval(() => {

    const h = document.createElement('div');

    h.className = 'heart-bg';

    h.textContent = hearts[Math.floor(Math.random() * hearts.length)];

    h.style.left = Math.random() * 100 + 'vw';

    h.style.animationDuration = (4 + Math.random() * 4) + 's';

    h.style.fontSize = (18 + Math.random() * 20) + 'px';

    document.body.appendChild(h);

    setTimeout(() => h.remove(), 8000);

  }, 500);

</script>
